        <div class="transcript" id="transcript" aria-live="off"></div>
      </main>
      <footer>
        <div class="connection" id="connection" role="status" aria-live="polite" data-live="false">Connecting</div>

        <div class="status" id="status" role="alert" hidden></div>
        <button class="primary" id="start" disabled>Talk</button>
        <div class="controls">
          <button class="secondary" id="mute">Mute</button>
          <button class="secondary end" id="end">End</button>

    <script type="module">
      import { createFridayFace, createConversationCloser, attachTiltGaze } from "/face.js";
      (() => {
        const face = createFridayFace(document.getElementById("fridayFace"));
        const tg = window.Telegram?.WebApp;
        let stopTilt = () => {};
        tg?.ready();
        tg?.expand();
        tg?.setHeaderColor?.("#140b08");
        tg?.setBackgroundColor?.("#140b08");
        tg?.setBottomBarColor?.("#140b08");

        const SESSION_KEY = "agent:friday:main:thread:397420856:353435";
        const MODEL = "gpt-live-1-codex";
        const app = document.getElementById("app");
        const startButton = document.getElementById("start");
        const muteButton = document.getElementById("mute");
        const endButton = document.getElementById("end");
        const statusNode = document.getElementById("status");
        const connectionNode = document.getElementById("connection");
        const transcriptNode = document.getElementById("transcript");
        let captionTimer;
        const remoteAudio = document.getElementById("remote");

        // Use the original supported capture path. Experimental media mode was
        // silent on the user's Android WebView; ignore its persisted preference.
        try { localStorage.removeItem("friday.audioMode"); } catch {}
        function microphoneConstraints() {
          return { echoCancellation: true, noiseSuppression: true, autoGainControl: true };
        }

        const enableAudio = document.getElementById("enableAudio");
        let playRejected = false;
        let audioDiagnosticTimer;
        async function playRemote(input) {
          remoteAudio.srcObject = input;
          remoteAudio.muted = false;
          remoteAudio.volume = 1;
          try {
            await remoteAudio.play();
            playRejected = false;
            enableAudio.hidden = true;
          } catch {
            playRejected = true;
            enableAudio.hidden = false;
            status("Tap Enable sound to hear Friday", true);
          }
        }

        async function reportAudioHealth() {
          if (!peer || socket?.readyState !== WebSocket.OPEN) return;
          try {
            const stats = await peer.getStats();
            const inbound = [...stats.values()].find(item => item.type === "inbound-rtp" && item.kind === "audio");
            socket.send(JSON.stringify({ type: "audio.diagnostic", report: {
              packetsReceived: inbound?.packetsReceived || 0,
              bytesReceived: inbound?.bytesReceived || 0,
              totalAudioEnergy: inbound?.totalAudioEnergy || 0,
              currentTime: remoteAudio.currentTime,
              readyState: remoteAudio.readyState,
              paused: remoteAudio.paused,
              trackMuted: remoteAudio.srcObject?.getAudioTracks()[0]?.muted ?? true,
              playRejected,
            }}));
          } catch {}
        }

        function stopPlayback() {
          clearInterval(audioDiagnosticTimer);
          remoteAudio.pause();
          remoteAudio.srcObject = null;
          enableAudio.hidden = true;
        }

        enableAudio.addEventListener("click", async () => {
          try {
            await remoteAudio.play();
            playRejected = false;
            enableAudio.hidden = true;
            status("Live · you can interrupt anytime");
          } catch {
            status("Audio playback blocked. Close and reopen the Mini App.", true);
          }
        });

        let socket;
        let bridgeConnected = false;
        let gatewayReady = false;
        let starting = false;
        let peer;
        let stream;
        let voiceSessionId;
        let muted = false;
        let requestId = 0;
        const pending = new Map();

        const closer = createConversationCloser({
          close: () => endCall("Conversation ended"),
          pendingChanged: (pending) => status(pending ? "Ending conversation… speak to keep talking" : "Live · you can interrupt anytime"),
        });
        let closeMonitor;
        let closeMonitorBusy = false;
        let previousCloseEnergy = 0, previousCloseDuration = 0;
        async function monitorClosePlayback() {
          if (!closer.pending || !peer || closeMonitorBusy) return;
          closeMonitorBusy = true;
          const currentPeer = peer;
          try {
            const stats = await currentPeer.getStats();
            if (peer !== currentPeer || !closer.pending) return;
            const audio = [...stats.values()].find(s => s.type === "inbound-rtp" && (s.kind === "audio" || s.mediaType === "audio"));
            let level = NaN;
            if (audio) {
              const duration = audio.totalSamplesDuration - previousCloseDuration;
              level = Number.isFinite(audio.audioLevel) ? audio.audioLevel : duration > 0 ? Math.sqrt(Math.max(0, (audio.totalAudioEnergy - previousCloseEnergy) / duration)) : NaN;
              previousCloseEnergy = audio.totalAudioEnergy || 0;
              previousCloseDuration = audio.totalSamplesDuration || 0;
            }
            // Already stopped/blocked playback cannot have a farewell to drain.
            if (remoteAudio.paused || playRejected) level = 0;
            closer.sample(level);
          } catch { closer.sample(NaN); }
          finally { closeMonitorBusy = false; }
        }

        function showCaption(text) {
          clearTimeout(captionTimer);
          transcriptNode.textContent = String(text).trim().split(/\s+/).slice(-8).join(" ");
          transcriptNode.classList.add("visible");
          captionTimer = setTimeout(() => {
            transcriptNode.classList.remove("visible");
            captionTimer = setTimeout(() => { transcriptNode.textContent = ""; }, 650);
          }, 2400);
        }
        function clearCaption() {
          clearTimeout(captionTimer);
          transcriptNode.classList.remove("visible");
          transcriptNode.textContent = "";
        }
        function updateConnection() {
          const ready = socket?.readyState === WebSocket.OPEN && bridgeConnected && gatewayReady;
          const live = Boolean(ready && voiceSessionId && peer?.connectionState === "connected");
          connectionNode.dataset.live = String(live);
          connectionNode.textContent = live ? "Live" : !ready ? "Offline" : starting || voiceSessionId ? "Connecting" : "Ready";
        }

        function status(text, error = false) {
          statusNode.textContent = text;
          statusNode.hidden = !error;
          updateConnection();
        }

        function updateReady() {
          const insideTelegram = Boolean(tg?.initData);
          startButton.disabled = starting || !insideTelegram || !bridgeConnected || !gatewayReady;
          startButton.textContent = "Talk";
          status(!insideTelegram ? "Open this Mini App from Friday in Telegram." : "", !insideTelegram);

        }

        function connect() {
          const protocol = location.protocol === "https:" ? "wss:" : "ws:";
          socket = new WebSocket(`${protocol}//${location.host}/rpc`);
          socket.addEventListener("open", () => socket.send(JSON.stringify({ type: "auth", initData: tg?.initData || "" })));
          socket.addEventListener("message", async (event) => {
            const message = JSON.parse(event.data);
            if (message.type === "auth.ok") {
              bridgeConnected = Boolean(message.bridgeConnected);
              gatewayReady = Boolean(message.gatewayReady);
              updateReady();
              return;
            }
            if (message.type === "auth.error") {
              status(message.message || "Telegram authorization failed", true);
              return;
            }
            if (message.type === "bridge.state") {
              bridgeConnected = Boolean(message.connected);
              gatewayReady = Boolean(message.gatewayReady);
              updateReady();
              return;
            }
            if ((message.type === "result" || message.type === "error") && pending.has(message.id)) {
              const task = pending.get(message.id);
              pending.delete(message.id);
              message.type === "result" ? task.resolve(message.result) : task.reject(new Error(message.message || "Request failed"));
              return;
            }
            if (message.type === "event" && message.event === "talk.event") handleTalkEvent(message.payload);
          });
          socket.addEventListener("close", () => {
            bridgeConnected = false;
            gatewayReady = false;
            updateReady();
            if (app.classList.contains("live")) endCall("Connection lost", true);
            setTimeout(connect, 1800);
          });
        }

        function request(type, body) {
          return new Promise((resolve, reject) => {
            const id = `r${++requestId}`;
            pending.set(id, { resolve, reject });
            socket.send(JSON.stringify({ type, id, ...body }));
            setTimeout(() => {
              if (!pending.has(id)) return;
              pending.delete(id);
              reject(new Error("Friday took too long to respond"));
            }, 32000);
          });
        }

        function rpc(method, params = {}) { return request("rpc", { method, params }); }

        function eventText(event) {
          const payload = event?.payload || {};
          return payload.text || payload.transcript || payload.delta || payload.message || "";
        }

        function handleTalkEvent(raw) {
          const event = raw?.talkEvent || raw;
          if (!event || !voiceSessionId || event.sessionId !== voiceSessionId) return;
          const text = eventText(event);
          face.event(event.type, text);
          if (["transcript.delta", "transcript.done", "output.text.delta", "output.text.done"].includes(event.type) && text) showCaption(text);
          if (event.type === "transcript.delta" || event.type === "transcript.done") closer.cancel();
          if (event.type === "output.text.delta" || event.type === "output.text.done") closer.output();
          if (event.type === "tool.result" && event.payload?.toolName === "end_conversation" &&
              event.payload?.source === "gpt-live-delegation" && event.payload?.status === "close_requested") {
            closer.request(event.payload.requestId);
          }
          if (event.type === "session.error") {
            status(text || "The voice session reported an error", true);
          } else if (event.type === "session.closed") {
            endCall("Talk ended");
            return;
          }
        }

        function waitForIce(peerConnection) {
          if (peerConnection.iceGatheringState === "complete") return Promise.resolve();
          return new Promise((resolve) => {
            const timeout = setTimeout(resolve, 2500);
            peerConnection.addEventListener("icegatheringstatechange", () => {
              if (peerConnection.iceGatheringState === "complete") {
                clearTimeout(timeout);
                resolve();
              }
            });
          });
        }

        function beginMeter(input) {
          const audioContext = new AudioContext();
          const analyser = audioContext.createAnalyser();
          analyser.fftSize = 256;
          audioContext.createMediaStreamSource(input).connect(analyser);
          const data = new Uint8Array(analyser.frequencyBinCount);
          let speechSince = 0;
          const tick = () => {
            if (!stream) { audioContext.close(); return; }
            analyser.getByteFrequencyData(data);
            const level = Math.min(1, data.reduce((sum, value) => sum + value, 0) / data.length / 90);
            document.documentElement.style.setProperty("--level", level.toFixed(3));
            if (level > .18 && !muted) {
              speechSince ||= performance.now();
              if (performance.now() - speechSince > 180) closer.cancel();
            } else speechSince = 0;
            requestAnimationFrame(tick);
          };
          tick();
        }

        async function startCall() {
          starting = true;
          startButton.disabled = true;
          status("Waiting for microphone permission…");
          try {
            stream = await navigator.mediaDevices.getUserMedia({
              audio: microphoneConstraints(),
            });
            beginMeter(stream);
            status("Opening GPT‑Live…");
            const session = await rpc("talk.client.create", {
              sessionKey: SESSION_KEY,
              provider: "openai",
              model: MODEL,
              mode: "realtime",
              transport: "webrtc",
              brain: "agent-consult",
              capabilities: ["gateway-control-v1"],
            });
            if (session.clientControl?.owner !== "gateway") throw new Error("Secure Gateway control was not negotiated");
            voiceSessionId = session.voiceSessionId;

            peer = new RTCPeerConnection();
            stream.getTracks().forEach((track) => peer.addTrack(track, stream));
            peer.addEventListener("track", (event) => {
              void playRemote(event.streams[0] || new MediaStream([event.track]));
            });
            peer.addEventListener("connectionstatechange", () => {
              updateConnection();
              if (peer?.connectionState === "connected") status("Live · you can interrupt anytime");
              if (["failed", "closed"].includes(peer?.connectionState)) endCall("Voice connection ended", true);
            });
            const offer = await peer.createOffer({ offerToReceiveAudio: true });
            await peer.setLocalDescription(offer);
            await waitForIce(peer);
            const answer = await request("offer", {
              offerUrl: session.offerUrl,
              offerHeaders: session.offerHeaders || {},
              clientSecret: session.clientSecret,
              sdp: peer.localDescription.sdp,
            });
            await peer.setRemoteDescription({ type: "answer", sdp: answer.sdp });
            app.classList.add("live");
            face.start(peer, remoteAudio);
            audioDiagnosticTimer = setInterval(reportAudioHealth, 5000);
            closeMonitor = setInterval(monitorClosePlayback, 120);
            updateConnection();
            tg?.HapticFeedback?.impactOccurred?.("medium");
          } catch (error) {
            const message = error?.message || "Could not start voice";
            await endCall(message, true);
          } finally {
            starting = false;
            startButton.disabled = !tg?.initData || !bridgeConnected || !gatewayReady;
            updateConnection();
          }
        }

        async function endCall(message = "Ready when you are", error = false) {
          face.stop();
          clearCaption();
          closer.reset();
          clearInterval(closeMonitor);
          previousCloseEnergy = 0; previousCloseDuration = 0;
          starting = false;
          const closingSession = voiceSessionId;
          voiceSessionId = undefined;
          try { peer?.close(); } catch {}
          peer = undefined;
          stream?.getTracks().forEach((track) => track.stop());
          stream = undefined;
          stopPlayback();
          document.documentElement.style.setProperty("--level", "0");
          app.classList.remove("live");
          muted = false;
          muteButton.textContent = "Mute";
          if (closingSession && socket?.readyState === WebSocket.OPEN) {
            rpc("talk.client.close", { sessionKey: SESSION_KEY, voiceSessionId: closingSession }).catch(() => {});
          }
          updateReady();
          status(message, error);
        }

        startButton.addEventListener("click", startCall);
        muteButton.addEventListener("click", () => {
          muted = !muted;
          stream?.getAudioTracks().forEach((track) => { track.enabled = !muted; });
          muteButton.textContent = muted ? "Unmute" : "Mute";
          tg?.HapticFeedback?.selectionChanged?.();
        });
        endButton.addEventListener("click", () => endCall());
        window.addEventListener("pagehide", () => endCall("Talk ended"));
        window.addEventListener("pagehide", () => { stopTilt(); face.destroy(); }, { once: true });
        connect();
        updateReady();
        try { stopTilt = attachTiltGaze(face, tg); } catch { /* Optional motion must never block voice. */ }
      })();
    </script>
